Switch0
// Do this on other switches.
// Same VLAN numbers must be used wherever needed.
// Each VLAN has a separate VLAN number.

en
conf t

vlan 10
name ITLAB
ex

vlan 20
name COMPLAB
ex

// Assign ports to VLAN 10
int range fa0/1-2
switchport mode access
switchport access vlan 10
ex

// Assign ports to VLAN 20
int range fa0/3-4
switchport mode access
switchport access vlan 20
ex

// Trunk port connected to the router
int fa0/5
switchport mode trunk
ex

end
wr

Router
en
conf t

// Enable the physical interface connected to the switch
int gig0/0
no shut
ex

// VLAN 10 - ITLAB
int gig0/0.10
encapsulation dot1Q 10
ip add 192.168.1.1 255.255.255.0
ex

// VLAN 20 - COMPLAB
int gig0/0.20
encapsulation dot1Q 20
ip add 192.168.2.1 255.255.255.0
ex


// DHCP for VLAN 10
ip dhcp excluded-address 192.168.1.1 192.168.1.10

ip dhcp pool ITLAB
network 192.168.1.0 255.255.255.0
default-router 192.168.1.1
ex


// DHCP for VLAN 20
ip dhcp excluded-address 192.168.2.1 192.168.2.10

ip dhcp pool COMPLAB
network 192.168.2.0 255.255.255.0
default-router 192.168.2.1
ex

// Turn off the router before adding a serial module.
// Install the required serial module, then turn it on.
// Do the same with the other routers.
// Every router must have a unique IP address on each link.
// The two ends of each serial link must be in the same subnet.


// Serial connection 1
int se0/3/0
ip add 10.0.0.1 255.255.255.252
no shut
// Set clock rate only if this interface is the DCE end.
clock rate 64000
ex


// Serial connection 2
int se0/3/1
ip add 20.0.0.1 255.255.255.252
no shut
// Set clock rate only if this interface is the DCE end.
clock rate 64000
ex


// Serial connection 3
int se0/2/1
ip add 30.0.0.1 255.255.255.252
no shut
// Set clock rate only if this interface is the DCE end.
clock rate 64000
ex

// OSPF configuration
// Every router must have a unique router ID.
// Router 1 = 1.1.1.1
// Router 2 = 2.2.2.2
// Router 3 = 3.3.3.3
// Router 4 = 4.4.4.4

router ospf 1
router-id 1.1.1.1

// Advertise VLAN 10
network 192.168.1.0 0.0.0.255 area 0

// Advertise VLAN 20
network 192.168.2.0 0.0.0.255 area 0

// Advertise serial connections
network 10.0.0.0 0.0.0.3 area 0
network 20.0.0.0 0.0.0.3 area 0
network 30.0.0.0 0.0.0.3 area 0

ex

end
